Har bir belgining qiymatini hash map'dan olamiz. Agar belgi o'ngdagi belgidan kichik bo'lsa (IV, IX, XC), u ayiriladi, aks holda qo'shiladi. Satr chapdan o'ngga bir marta aylanib chiqiladi. Map'da atigi 7 ta kalit bor, shuning uchun xotira doimiy.

class Solution:
    def romanToInt(self, s: str) -> int:
        roman = {
            'I': 1,
            'V': 5,
            'X': 10,
            'L': 50,
            'C': 100,
            'D': 500,
            'M': 1000
        }

        total = 0
        n = len(s)

        for i in range(n):
            if i < n - 1 and roman[s[i]] < roman[s[i + 1]]:
                total -= roman[s[i]]
            else:
                total += roman[s[i]]

        return total
