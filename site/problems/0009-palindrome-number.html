Sonni stringga aylantirmaymiz. Faqat yarmini teskari o'giramiz: x kichrayib, reversed_half o'sib borib, o'rtada uchrashadi. Shunda butun sonni teskari aylantirishdagi overflow xavfi ham yo'qoladi. Toq xonali sonda o'rtadagi raqam reversed_half // 10 bilan chetlab o'tiladi.

class Solution:
    def isPalindrome(self, x: int) -> bool:
        if x < 0 or (x % 10 == 0 and x != 0):
            return False

        reversed_half = 0

        while x > reversed_half:
            reversed_half = reversed_half * 10 + x % 10
            x //= 10

        return x == reversed_half or x == reversed_half // 10

class Solution {

    /**
     * @param Integer $x
     * @return Boolean
     */
    function isPalindrome($x) {
        if ($x < 0 || ($x % 10 == 0 && $x != 0)) {
            return false;
        }

        $reversedHalf = 0;

        while ($x > $reversedHalf) {
            $reversedHalf = $reversedHalf * 10 + $x % 10;
            $x = intdiv($x, 10);
        }

        return $x == $reversedHalf || $x == intdiv($reversedHalf, 10);
    }
}
